\makepart{选择题}{本题满分10分，每小题2分}


\begin{problem}
函数\pickin{}在其定义域内连续．
\begin{abcd}
\item $f(x) = \ln x + \sin x$
\item $f(x) = \begin{cases}
  \sin x, & x \le 0, \\
  \cos x, & x > 0 \\
\end{cases}$
\item $f(x) = \begin{cases}
  x + 1, &  x < 0, \\
      0, &  x = 0, \\
  x - 1, &  x > 0 \\
\end{cases}$
\item $f(x) = \begin{cases}
  \frac{1}{\sqrt{|x|}}, & x \ne 0, \\
                     0, & x = 0 \\
\end{cases}$
\end{abcd}
\end{problem}


\begin{problem}
若函数$f(x)$在区间$(a,b)$内可导，$x_1$和$x_2$是区间$(a,b)$内任意两点，
且$x_1 < x_2$，则至少存在一点$\xi$，使\pickout{}
\begin{abcd}
\item $f(b) - f(a) = f'(\xi)(b - a)$，其中$a < \xi < b$．
\item $f(b) - f(x_1) = f'(\xi)(b - x_1)$，其中$x_1 < \xi < b$．
\item $f(x_2) - f(x_1) = f'(\xi)(x_2 - x_1)$，其中$x_1 < \xi < x_2$．
\item $f(x_2) - f(a) = f'(\xi)(x_2 - a)$，其中$a < \xi < x_2$．
\end{abcd}
\end{problem}


\begin{problem}
广义积分\pickin{}收敛．
\begin{abcd}
\item $\int_{\e}^{+\infty} \frac{\ln x}{x}\dx$．
\item $\int_{\e}^{+\infty} \frac{\dx}{x\ln x}$．
\item $\int_{\e}^{+\infty} \frac{\dx}{x(\ln x)^2}$．
\item $\int_{\e}^{+\infty} \frac{\dx}{x\sqrt{\ln x}}$．
\end{abcd}
\end{problem}


\begin{problem}
假设$A$是$n$阶方阵，其秩$r<n$，那么在$A$的$n$个行向量中\pickout{}
\begin{abcd}
\item 必有$r$个行向量线性无关．
\item 任意$r$个行向量都线性无关．
\item 任意$r$个行向量都构成极大线性无关向量组．
\item 任意$r$个行向量都可以由其他$r$个行向量线性表示．
\end{abcd}
\end{problem}


\begin{problem}
若二事件$A$和$B$同时出现的概率$P(AB)=0$，则\pickout{}
\begin{abcd}
\item $A$和$B$不相容（相斥）．
\item $AB$是不可能事件．
\item $AB$未必是不可能事件．
\item $P(A) = 0$或$P(B) = 0$．
\end{abcd}
\end{problem}
