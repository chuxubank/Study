\begin{problem}
若曲线$y = x^2 + ax + b$和$2y = - 1 + xy^3$在点$(1, -1)$处相切，其中$a$, $b$是常数，则\pickout{}
\begin{abcd}
\item $a = 0,b = - 2$．		
\item $a = 1,b = - 3$．

\end{abcd}

\begin{problem}
设函数$f(x) = \begin{cases}
    x^2, & 0 \le x \le 1, \\
  2 - x, & 1 < x \le 2.
\end{cases}$
记$F(x) = \int_0^xf (t)\dt$, $0 \le x \le 2$，则\pickout{}
\begin{abcd}
\item $F(x) = \begin{cases}
                     \frac{x^3}{3}, & 0 \le x \le 1; \\
  \frac{1}{3} + 2x - \frac{x^2}{3}, & 1 < x \le 2.
\end{cases}$
\item $F(x) = \begin{cases}
                      \frac{x^3}{3}, & 0 \le x \le 1; \\
  -\frac{7}{6} + 2x - \frac{x^2}{2}, & 1 < x \le 2.
\end{cases}$
\item $F(x) = \begin{cases}
                       \frac{x^3}{3}, & 0 \le x \le 1; \\
  \frac{x^2}{3} + 2x - \frac{x^2}{2}, & 1 < x \le 2.
\end{cases}$
\item $F(x) = \begin{cases}
       \frac{x^3}{3}, & 0 \le x \le 1; \\
  2x - \frac{x^2}{2}, & 1 < x \le 2.
\end{cases}$
\end{abcd}
\end{problem}


\begin{problem}
设函数$f(x)$在$(-\infty , +\infty)$内有定义，$x_0 \ne 0$是函数$f(x)$的极大值点，则\pickout{}
\begin{abcd}
\item $x_0$必是$f(x)$的驻点					
\item $-x_0$必是$-f(-x)$的极小值点
\item $-x_0$必是$-f(x)$的极小值点			
\item 对一切$x$都有$f(x) \le f(x_0)$
\end{abcd}
\end{problem}
