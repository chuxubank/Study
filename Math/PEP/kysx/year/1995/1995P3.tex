\begin{problem}
设$f(x)$和$\phi (x)$在$(-\infty, +\infty)$上有定义，$f(x)$为连续函数，
且$f(x) \ne 0$，$\phi (x)$有间断点，则\pickout{}
\begin{abcd}
\item $\phi [f(x)]$必有间断点．		
\item $[\phi(x)]^2$必有间断点．
\item $f[\phi(x)]$必有间断点．		
\item $\frac{\phi(x)}{f(x)}$必有间断点．
\end{abcd}
\end{problem}


\begin{problem}
曲线$y = x(x - 1)(2 - x)$与$x$轴所围图形的面积可表示为\pickout{}
\begin{abcd}
\item $- \int_0^2x(x - 1)(2 - x)\dx$．
\item $\int_0^1x(x - 1)(2 - x)\dx - \int_1^2x(x - 1)(2 - x)\dx$．
\item $- \int_0^1x(x - 1)(2 - x)\dx + \int_1^2x(x - 1)(2 - x)\dx$．
\item $\int_0^2x(x - 1)(2 - x)\dx$．
\end{abcd}
\end{problem}


\begin{problem}
设$f(x)$在$(-\infty , +\infty)$内可导，且对任意$x_1$, $x_2$，
当$x_1 > x_2$时都有$f(x_1) > f(x_2)$，则\pickout{}
\begin{abcd}
\item 对任意$x$，$f'(x) > 0$．		
\item 对任意$x$，$f'(- x) \le 0$．
\item 函数$f(- x)$单调增加．
\item 函数$- f(- x)$单调增加．
\end{abcd}
\end{problem}
