\begin{problem}
设$x \to 0$时$\e^x - (ax^2 + bx + 1)$是比$x^2$高阶的无穷小，则\pickout{}
\begin{abcd}
\item $a = \frac{1}{2},b = 1$．		
\item $a = 1,b = 1$．
\item $a = - \frac{1}{2},b = - 1$．		
\item $a = - 1,b = 1$．
\end{abcd}
\end{problem}


\begin{problem}
设函数$f(x)$在区间$(- \delta ,\delta)$内有定义，
若当$x \in(- \delta ,\delta)$时，恒有$\left| f(x) \right| \le x^2$，
则$x = 0$必是$f(x)$的\pickout{}
\begin{abcd}
\item 间断点． 							
\item 连续而不可导的点．
\item 可导的点，且$f'(0) = 0$．			
\item 可导的点，且$f'(0) \ne 0$．
\end{abcd}
\end{problem}


\begin{problem}
设$f(x)$处处可导，则\pickout{}
\begin{abcd}
\item 当$\lim_{x \to -\infty} f(x) = -\infty$时，
      必有$\lim_{x \to -\infty} f'(x) = -\infty$．
\item 当$\lim_{x \to -\infty} f'(x) = -\infty$时，
      必有$\lim_{x \to -\infty} f(x) = -\infty$．
\item 当$\lim_{x \to +\infty} f(x) = +\infty$时，
      必有$\lim_{x \to +\infty} f'(x) = +\infty$．
\item 当$\lim_{x \to +\infty} f'(x) = +\infty$时，
      必有$\lim_{x \to +\infty} f(x) = +\infty$．
\end{abcd}
\end{problem}

\item 有无穷多个实根．

\begin{problem}
设$f(x)$,$g(x)$在区间$[a,b]$上连续，$g(x) < f(x) < m$（$m$为常数），
由曲线$y = g(x)$, $y = f(x)$, $x = a$及$x = b$所围平面图形绕直线$y = m$旋转而成的旋转体体积为\pickout{}
\begin{abcd}
\item $\int_a^b \pi[2m - f(x) + g(x)][f(x) - g(x)]\dx$．
\item $\int_a^b \pi[2m - f(x) - g(x)][f(x) - g(x)]\dx$．
\item $\int_a^b \pi[m - f(x) + g(x)][f(x) - g(x)]\dx$．
\item $\int_a^b \pi[m - f(x) - g(x)][f(x) - g(x)]\dx$．
\end{abcd}
\end{problem}
