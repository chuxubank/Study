\begin{problem}
已知函数$y = f(x)$对一切$x$满足$xf''(x) + 3x[f'(x)]^2 = 1 - \e^{- x}$，
若$f'(x_0) = 0$（$x_0 \ne 0$），则\pickout{}
\begin{abcd}
\item $f(x_0)$是$f(x)$的极大值．
\item $f(x_0)$是$f(x)$的极小值．
\item $(x_0,f(x_0))$是曲线$y = f(x)$的拐点．
\item $f(x_0)$不是$f(x)$的极小值，$(x_0,f(x_0))$也不是曲线$y = f(x)$的拐点．
\end{abcd}
\end{problem}
