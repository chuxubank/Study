\item 在$(a,b)$内存在点$\xi$，使$\frac{b^2 - a^2}{\int_a^b f(x)\dx} = \frac{2\xi}{f(\xi)}$；
\item 在$(a,b)$内存在与(2)中$\xi$相异的点$\eta$，使
      $f'(\eta)(b^2 - a^2) = \smash[t]{\frac{2\xi}{\xi - a}\int_a^b f(x)\dx}$．
\end{enumerate}
\end{problem}
