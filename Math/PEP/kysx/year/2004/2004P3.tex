\begin{problem}[points=8]
设$f(x)$, $g(x)$在$[a,b]$上连续，且满足
$$\int_{a}^{x} f(t)\dt \ge \int_{a}^{x} g(t)\dt,\;\;x \in[a,b),
  \qquad \int_{a}^{b} f(t)\dt = \int_{a}^{b} g(t)\dt.$$
证明：$\int_{a}^{b} xf(x)\dx \le \int_{a}^{b} xg(x)\dx$．
\end{problem}
