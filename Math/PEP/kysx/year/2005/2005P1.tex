\begin{problem}
设函数$f(x) = \lim_{n \to \infty} \sqrt[n]{1+|x|^{3n}}$，则$f(x)$在$(-\infty , +\infty)$内\pickout{}
\begin{abcd}
\item 处处可导．
\item 恰有一个不可导点．
\item 恰有两个不可导点．
\item 至少有三个不可导点．
\end{abcd}
\end{problem}


\begin{problem}
设$F(x)$是连续函数$f(x)$的一个原函数，表示“$M$的充分必要条件是$N$”，则必有\pickout{}
\begin{abcd}
\item $F(x)$是偶函数$\Leftrightarrow$$f(x)$是奇函数．
\item $F(x)$是奇函数$\Leftrightarrow$$f(x)$是偶函数．
\item $F(x)$是周期函数$\Leftrightarrow$$f(x)$是周期函数．
\item $F(x)$是单调函数$\Leftrightarrow$$f(x)$是单调函数．
\end{abcd}
\end{problem}


\begin{problem}
设函数$u(x,y) = \phi(x + y) + \phi(x - y) + \int_{x - y}^{x + y} \psi(t)\dt$，
其中函数$\phi$具有二阶导数，$\psi$具有一阶导数，则必有\pickout{}
\begin{abcd}
\item $\frac{\pd^2 u}{\pd x^2} = - \frac{\pd^2 u}{\pd y^2}$．
\item $\frac{\pd^2 u}{\pd x^2} = \frac{\pd^2 u}{\pd y^2}$．
\item $\frac{\pd^2 u}{\pdx\pd y} = \frac{\pd^2 u}{\pd y^2}$．
\item $\frac{\pd^2 u}{\pdx\pd y} = \frac{\pd^2 u}{\pd x^2}$．
\end{abcd}
\end{problem}


\begin{problem}
设有三元方程$xy - z\ln y + \e^{xz} = 1$，根据隐函数存在定理，
存在点$(0,1,1)$的一个邻域，在此邻域内该方程\pickout{}
\begin{abcd}
\item 只能确定一个具有连续偏导数的隐函数$z = z(x,y)$．
\item 可确定两个具有连续偏导数的隐函数和$y = y(x,z)$和$z = z(x,y)$．
\item 可确定两个具有连续偏导数的隐函数$x = x(y,z)$和$z = z(x,y)$．
\item 可确定两个具有连续偏导数的隐函数$x = x(y,z)$和$y = y(x,z)$．
\end{abcd}
\end{problem}
