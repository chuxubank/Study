\begin{problem}[points=13]
设二维随机变量$(X,Y)$的概率分布为
$$\begin{array}{|c|ccc|}
\hline
  \diagboxtwo{X}{Y} &  -1 &   0 & 1 \\
\hline
             -1 &   a &   0 & 0.2 \\
              0 & 0.1 &   b & 0.2 \\
              1 &   0 & 0.1 & c \\
\hline
\end{array}$$
其中$a$, $b$, $c$为常数，且$X$的数学期望$EX = - 0.2$，$P\{Y \le 0|X \le 0\} = 0.5$，
记$Z = X + Y$，求：\par
\begin{enumerate*}
\item $a,b,c$的值；
\item $Z$的概率分布；
\item $P\{X = Z\}$．
\end{enumerate*}
\end{problem}
