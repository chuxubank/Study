\begin{problem}[points=11]
设随机变量$X$与$Y$独立同分布，且$X$的概率分布为
$$\begin{array}{|c|cc|}
\hline
  X &   1 & 2 \\
\hline
  P & 2/3 & 1/3 \\
\hline
\end{array}$$
记$U = \max \{X,Y\}$, $V = \min \{X,Y\}$．
\begin{enumerate}
  \item 求$(U,V)$的概率分布；
  \item 求$U$与$V$的协方差$\Cov(U,V)$．
\end{enumerate}
\end{problem}
