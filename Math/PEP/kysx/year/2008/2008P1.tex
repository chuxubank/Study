\begin{problem}[points=12]
设$n$元线性方程组$Ax = b$，其中
$$A = \begin{pmatrix}
   2a &      1 &        & \\
  a^2 &     2a & \ddots & \\
      & \ddots & \ddots & 1 \\
      &        &    a^2 & 2a
\end{pmatrix},\quad x = \begin{pmatrix}
  x_1 \\
  x_2 \\
  \vdots \\
  x_n
\end{pmatrix}\quad b = \begin{pmatrix}
  1 \\
  0 \\
  \vdots \\
  0
\end{pmatrix}.$$
\begin{enumerate}
\item 求证$|A| = \left(n + 1 \right) a^n$；
\item $a$为何值，方程组有唯一解，求$x_1$；
\item $a$为何值，方程组有无穷多解，并求通解．
\end{enumerate}
\end{problem}

\begin{problem}[points=11]
设$X_1,X_2, \cdots ,X_n$是总体为$N(\mu ,\sigma^2)$的简单随机样本．记
$$\overline{X} = \frac{1}{n}\sum_{i = 1}^n X_i ,
  \quad S^2 = \frac{1}{n - 1}\sum_{i = 1}^n (X_i - \overline{X})^2 ,
  \quad T = \overline{X}^2 - \frac{1}{n} S^2.$$
\begin{enumerate*}
\item 证明 $T$是$\mu^2$的无偏估计量；
\item 当$\mu = 0,\sigma = 1$时，求$DT$．
\end{enumerate*}
\end{problem}
