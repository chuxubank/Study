\begin{problem}[points=11]

\begin{problem}[points=10]

\end{problem}


\begin{problem}[points=10]
设函数$f(x)$在闭区间$[0,1]$上连续，在开区间$(0,1)$内可导，
且$f(0) = 0$, $f(1) = \frac{1}{3}$．证明：存在$\xi \in \left(0,\frac{1}{2}\right)$,
$\eta \in \left(\frac{1}{2},1\right)$，使得$f'(\xi) + f'(\eta) = \xi^2 + \eta^2$．
