\begin{problem}
已知$X_1,X_2, \cdots ,X_n$为来自总体$X \sim N\left(\mu ,\sigma^2 \right)$的简单随机样本，
$$\overline{X} = \frac{1}{n}\sum_{i = 1}^n X_i,\quad
  S = \sqrt{\frac{1}{n - 1}\sum_{i = 1}^n \left(X_i - \overline{X} \right)^2},\quad
  S^* = \sqrt{\frac{1}{n - 1}\sum_{i = 1}^n \left(X_i - \mu \right)^2},$$
则\pickout{}
\begin{abcd}
\item $\frac{\sqrt{n} \left(\overline{X} - \mu \right)}{S} \sim t(n)$．
\item $\frac{\sqrt{n} \left(\overline{X} - \mu \right)}{S} \sim t(n - 1)$．
\item $\frac{\sqrt{n} \left(\overline{X} - \mu \right)}{S^*} \sim t(n)$．
\item $\frac{\sqrt{n} \left(\overline{X} - \mu \right)}{S^*} \sim t(n - 1)$．
\end{abcd}
\end{problem}
